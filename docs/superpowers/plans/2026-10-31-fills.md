**Goal:** Per maat een aanpasbare blazersfill (nootnamen) op de songpagina, geschreven voor het gekozen instrument.

**Architecture:** Noten zijn intern concert `[pc, letter]` (letter-gebaseerd transponeren zoals de akkoorden, `core/chords.py`). Opslag in een eigen tabel `fill` los van `bar`. Het schema-antwoord geeft per maat `fill` (geschreven namen); `PUT /api/songs/{id}/fills/{nr}` bewaart wat Bob typt. De songpagina toont een `♪`-regel per maatvak en heeft een bewerkstand met `prompt()`.

**Tech Stack:** Python 3 stdlib (`core/`), FastAPI + sqlite3 (`app/`), vanilla JS (Node v12-compatibel), pytest + node-harness.

**Spec:** `docs/superpowers/specs/2026-10-31-fills-design.md`

# --- fills (blazerspartij per maat) ----------------------------------------
MAX_FILL = 32
_FILL_SCHEIDING = re.compile(r"[\s,\-–]+")


def parse_fill(tekst: str, voice: str) -> list:
    """Fill zoals Bob hem typt, geschreven voor `voice` -> concert [[pc, letter], ...].

    Scheiding: spaties, komma's, '-' of '–'. Een noot = letter A-G (kleine letter mag)
    + hooguit één voorteken (#, ♯, b, ♭). Lege tekst -> [].
    """
    _check_voice(voice)
    noten = []
    for woord in _FILL_SCHEIDING.split(tekst.strip()):
        if not woord:
            continue
        try:
            pc, letter, rest = _parse_note(woord[0].upper() + woord[1:])
        except ValueError:
            rest = None
        if rest != "":
            raise ValueError(f"Onbekende noot: {woord!r}.")
        noten.append([(pc - VOICES[voice]) % 12, (letter - VOICE_LETTERS[voice]) % 7])
    if len(noten) > MAX_FILL:
        raise ValueError(f"Te veel noten (max. {MAX_FILL}).")
    return noten


def fill_names(noten: list, voice: str) -> list:
    """Concert [[pc, letter], ...] -> geschreven nootnamen voor `voice`."""
    return [note_name(*transpose_note(pc, letter, voice)) for pc, letter in noten]
```

def zet_fill(conn, song_id: int, nr: int, noten: list) -> bool:
    """Blazersfill van één maat (concert [[pc, letter], ...]); [] wist hem.

    Los van de tabel bar, zodat een controle (alle maten opnieuw) de fills laat staan.
    False als het nummer niet bestaat.
    """
    with conn:
        if not conn.execute("SELECT 1 FROM song WHERE id = ?", (song_id,)).fetchone():
            return False
        if noten:
            conn.execute("INSERT OR REPLACE INTO fill (song_id, nr, noten) VALUES (?,?,?)",
                         (song_id, nr, json.dumps(noten)))
        else:
            conn.execute("DELETE FROM fill WHERE song_id = ? AND nr = ?", (song_id, nr))
    return True


def fills(conn, song_id: int) -> dict:
    """{maatnummer: [[pc, letter], ...]} (concert) voor een nummer."""
    return {r["nr"]: json.loads(r["noten"])
            for r in conn.execute("SELECT nr, noten FROM fill WHERE song_id = ?", (song_id,))}
```

function maatRaster(maten, opTik, fillRegels = {}) {

    fillRegels[m.nr] = fill;
    return el("div", { class: "maatvak", "data-nr": m.nr, onclick: () => opTik(m.nr) },
      el("span", { class: "nr" }, m.nr),

  async function bewerkFill(nr) {
    const m = schema.delen.flatMap(d => d.maten).find(x => x.nr === nr);
    if (!m) return;
    const inst = schema.instrument;
    const tekst = prompt(`Fill maat ${nr} (geschreven voor ${INSTRUMENTEN[inst]}), bv. F E:`,
      (m.fill || []).join(" "));
    if (tekst === null) return;  // Annuleren
    try {
      const r = await api(`/api/songs/${id}/fills/${nr}`, { method: "PUT", json: { noten: tekst, instrument: inst } });
      m.fill = r.fill;
      if (fillRegels[nr]) fillRegels[nr].textContent = fillTekst(r.fill);
    } catch (e) {
      $("#hulp").textContent = `Fill niet bewaard: ${e.message}`;
    }
  }
